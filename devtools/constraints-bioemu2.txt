# pip constraints for installing AIMMD-Lab with the bioemu2 extras:
#
#   pip install -e ".[cgaimmd,bioemu2,graphs,tests]" -c devtools/constraints-bioemu2.txt
#
# bioemu2 0.1.0rc0 leaves most of its dependencies unpinned, and some of them
# change between releases in ways that matter to the C-GAIMMD BioEmu2
# adaptor: biotite builds the structures, RDKit orders and matches the ligand
# atoms, hydra and omegaconf read the model configuration. These are the
# versions the test lanes of the C-GAIMMD branch ran with (Python 3.13,
# torch 2.11). A constraint only bounds a package that something else asks
# for; it installs nothing by itself.
#
# Not pinned here, on purpose:
# - torch: chosen by the install route (the PyTorch CUDA or CPU index, or
#   conda-forge); bioemu2 itself asks for >= 2.11, < 2.14.
# - triton: every CUDA build of torch pins its own triton (torch 2.11: 3.6.0).
#
# The weekly bioemu2 canary job installs bioemu2 from upstream without this
# file, to see the newest versions early.
bioemu2==0.1.0rc0
biotite==2.0.0
biotraj==1.2.2
rdkit==2026.9.1
hydra-core==1.3.8
omegaconf==2.3.1
openmm==8.6.1
numpy>=2.1,<2.5
